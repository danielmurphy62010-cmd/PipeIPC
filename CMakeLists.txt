cmake_minimum_required(VERSION 3.24)
project(QuickIPC VERSION 0.1.0 LANGUAGES CXX)
if(NOT WIN32)
  message(FATAL_ERROR "QuickIPC currently supports Windows only.")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(blake3 CONFIG REQUIRED)
add_library(quickipc src/value.cpp src/protocol.cpp src/routing.cpp src/shared_vars.cpp src/pipe_transport.cpp)
target_include_directories(quickipc PUBLIC include)
target_link_libraries(quickipc PRIVATE blake3::blake3)
target_compile_definitions(quickipc PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
target_compile_options(quickipc PRIVATE /W4 /permissive-)
add_executable(quickipc_demo examples/demo.cpp)
target_link_libraries(quickipc_demo PRIVATE quickipc)
