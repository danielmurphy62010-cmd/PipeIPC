cmake_minimum_required(VERSION 3.24)
project(QuickIPC VERSION 1.0.0 LANGUAGES C CXX)
if(NOT WIN32)
  message(FATAL_ERROR "QuickIPC is Windows-only.")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
include(FetchContent)
FetchContent_Declare(blake3 GIT_REPOSITORY https://github.com/BLAKE3-team/BLAKE3.git GIT_TAG 1.5.5 GIT_SHALLOW TRUE SOURCE_SUBDIR c)
FetchContent_MakeAvailable(blake3)
add_library(quickipc src/value.cpp src/protocol.cpp src/routing.cpp src/shared_vars.cpp src/pipe.cpp src/runtime.cpp)
set_target_properties(quickipc PROPERTIES OUTPUT_NAME quickipc_core)
target_include_directories(quickipc PUBLIC include)
target_link_libraries(quickipc PRIVATE BLAKE3::blake3 bcrypt ole32)
target_compile_definitions(quickipc PUBLIC NOMINMAX WIN32_LEAN_AND_MEAN)
target_compile_options(quickipc PRIVATE /W4 /permissive- /EHsc)
add_executable(quickipc_server examples/server.cpp)
target_link_libraries(quickipc_server PRIVATE quickipc)
add_executable(quickipc_client examples/client.cpp)
target_link_libraries(quickipc_client PRIVATE quickipc)
option(QUICKIPC_BUILD_TESTS "Build unit tests" ON)
if(QUICKIPC_BUILD_TESTS)
 enable_testing()
 add_executable(quickipc_tests tests/tests.cpp)
 target_link_libraries(quickipc_tests PRIVATE quickipc)
 add_test(NAME quickipc_tests COMMAND quickipc_tests)
endif()
option(QUICKIPC_BUILD_PYTHON "Build the Python extension" OFF)
if(QUICKIPC_BUILD_PYTHON)
 find_package(Python 3.10 REQUIRED COMPONENTS Interpreter Development.Module)
 FetchContent_Declare(pybind11 GIT_REPOSITORY https://github.com/pybind/pybind11.git GIT_TAG v2.13.6 GIT_SHALLOW TRUE)
 FetchContent_MakeAvailable(pybind11)
 pybind11_add_module(quickipc_python MODULE bindings/python/module.cpp)
 set_target_properties(quickipc_python PROPERTIES OUTPUT_NAME quickipc)
 target_link_libraries(quickipc_python PRIVATE quickipc)
endif()
option(QUICKIPC_BUILD_JAVA "Build the Java JNI bridge" OFF)
if(QUICKIPC_BUILD_JAVA)
 find_package(JNI REQUIRED)
 add_library(quickipc_jni SHARED bindings/java/jni.cpp)
 target_include_directories(quickipc_jni PRIVATE ${JNI_INCLUDE_DIRS})
 target_link_libraries(quickipc_jni PRIVATE quickipc ${JNI_LIBRARIES})
 set_target_properties(quickipc_jni PROPERTIES OUTPUT_NAME quickipc_jni)
endif()
